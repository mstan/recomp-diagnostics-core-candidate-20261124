cmake_minimum_required(VERSION 3.20)
project(recomp_diagnostics_core LANGUAGES C CXX)
find_package(Threads REQUIRED)
add_library(recomp_diagnostics_core STATIC src/diagnostics_server.c src/replay_writer.cpp src/replay_c.cpp src/session.cpp src/owner_mailbox.cpp src/opaque_writer.cpp src/opaque_c.cpp)
add_library(recomp-diagnostics-core::core ALIAS recomp_diagnostics_core)
target_include_directories(recomp_diagnostics_core PUBLIC include)
target_compile_features(recomp_diagnostics_core PUBLIC c_std_11 cxx_std_17 PRIVATE cxx_std_20)
target_link_libraries(recomp_diagnostics_core PUBLIC Threads::Threads)
if(WIN32)
 target_link_libraries(recomp_diagnostics_core PUBLIC ws2_32)
endif()
include(CTest)
