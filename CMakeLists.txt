cmake_minimum_required(VERSION 3.20)
project(recomp_diagnostics_core LANGUAGES C CXX)
find_package(Threads REQUIRED)
add_library(recomp_diagnostics_core STATIC src/diagnostics_server.c src/replay_writer.cpp)
add_library(recomp-diagnostics-core::core ALIAS recomp_diagnostics_core)
target_include_directories(recomp_diagnostics_core PUBLIC include)
target_compile_features(recomp_diagnostics_core PUBLIC c_std_11 cxx_std_20)
target_link_libraries(recomp_diagnostics_core PUBLIC Threads::Threads)
if(WIN32)
 target_link_libraries(recomp_diagnostics_core PUBLIC ws2_32)
endif()
include(CTest)
if(BUILD_TESTING)
 add_executable(diagnostics_loopback_test tests/loopback_test.cpp)
 target_link_libraries(diagnostics_loopback_test PRIVATE recomp_diagnostics_core)
 add_test(NAME diagnostics_loopback COMMAND diagnostics_loopback_test)
 add_executable(diagnostics_replay_test tests/replay_test.cpp)
 target_link_libraries(diagnostics_replay_test PRIVATE recomp_diagnostics_core)
 add_test(NAME diagnostics_replay COMMAND diagnostics_replay_test)
endif()

if(BUILD_TESTING)
 add_executable(diagnostics_replay_bounds tests/replay_bounds.cpp)
 target_link_libraries(diagnostics_replay_bounds PRIVATE recomp_diagnostics_core)
 add_test(NAME diagnostics_replay_bounds COMMAND diagnostics_replay_bounds)
endif()
